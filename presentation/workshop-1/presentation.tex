\end{itemize}

def temp_lim(var, env, name):
    """Temperature limitation of any rate: none, linear or exponential."""
    T = env["temperature"]
    refTemp = Const("refTemp", range=(5, 10))
    theta = Const("theta", range=(1.06, 1.13))
    return ChooseProcess(
        Process(Algebraic(var, 1), name="none"),
        Process(Algebraic(var, T / refTemp), name="linear"),
        Process(Algebraic(var, theta ** (T - refTemp)), name="exponential"),
        name=name)

influences = [temp_lim(phyto[f"temp{r}Lim"], env, name=f"temp{r}Influence")
              for r in ["Growth", "Resp", "Mort", "Exc", "Sed"]]

    \begin{graphicbox}[boxrule=2pt] 
    Given 
    \begin{itemize}
        \item ODE $ x'(t) =  F( x(t), t, \theta)$
        \item  measurements $(t_i, x_\text{obs}(t_i) + \epsilon_i)$, $\epsilon_i \sim N(0, \sigma^2)$
    \end{itemize}

 \end{frame}
\begin{frame}{Integration using multiple shooting\footfullcite{aydogmus2020modifiedmultipleshootingalgorithm}}
    \begin{itemize}
        \item integration: pick $x(t_0)$, solve ODE for $x_\theta(t)$, min discrepancy $\mathcal{L}(x_\theta)$ on data
        \item problem: chaotic systems
        \item solution: integrate shorter intervals, join trajectories   
    \end{itemize}
\end{frame}
\begin{frame}{Integration using multiple shooting\footfullcite{aydogmus2020modifiedmultipleshootingalgorithm}}
        \begin{graphicbox}[boxrule=2pt, title=Multiple shooting] 
        find $\theta, s_i$ minimizing discrepancy on data
        $$ \sum_i \mathcal{L}(x_i), \quad \text{where } x_i'=F(x_i,t,\theta), x_i(t_i)=s_i$$
        subject to
        $$
        x_i(t_{i+1}) = s_{i+1} \quad \forall i=0,1, \dots, N_\text{subintervals}-2
        $$
    \end{graphicbox}
\end{frame}

\begin{frame}{Gradient matching for fully-observed systems}
    \begin{itemize}
        \item $\hat x(t)$ - interpolant of the data 
        \item interpolation: Gaussian processes\footfullcite{wenk2019fastgaussianprocessbased}, cubic splines\footfullcite{varah-splines}
        %\item efective derivation
    \end{itemize}
    \begin{graphicbox}[boxrule=2pt, title=Least squares formulation]
       Find $
        \operatorname{argmin}_\theta\|\hat x'(t) - F(\hat x(t), t, \theta)  \|
        $
    \end{graphicbox}
\end{frame}
\begin{frame}{Integral matching\footfullcite{integra-matching-reject}}
% \begin{itemize}
%     \item derivative are noisy $ \propto \frac{1}{|t_{i+1} - t_i|}$
% \end{itemize}
Derivatives are noisy $ \propto \frac{1}{|t_{i+1} - t_i|}$!
    \begin{graphicbox}[boxrule=2pt]
        Minimize
        $$
        %\mathcal{L}(\theta) = 
        \displaystyle\int_{t_0 + \Delta}^{t_1} \left\|\hat x(t) -\hat x(t - \Delta) - \int_{t - \Delta} ^t F(\hat x(\tau), \tau, \theta)\, d\tau  \right\| dt 
        $$
    \end{graphicbox}
\end{frame}
\begin{frame}{Collocation\footfullcite{magi, simultaneous}}
    \begin{itemize}
        \item $x\colon I \to \mathbb{R}^n$ partially observed
        \item choose $l$ \emph{collocation} points $(\tau_1, s_1), (\tau_2, s_2), \dots, (\tau_l, s_l) \in I\times \mathbb{R}^n$  
        \item $\hat x(t)$: interpolant of $(\tau_i, s_i)_{i}$, {\color{gray} hyperparameters $\phi$}
        \item choose $s=(s_1, \dots, s_l)$ and $\phi$ subject to $\hat x(t)$ solves the ODE and matches the data 
    \end{itemize}
\end{frame}
\begin{frame}{Collocation}
%
     \begin{small}
        Let $\hat x(t)$ be the interpolant of $(\tau_i, s_i)$ with hyperparameters $\phi$ and collocation points $s=(s_1, \dots, s_l)$. Let $(t_i, y_i)$ be a set of noisy measurements.
     \end{small}
 %    
    \begin{graphicbox}
        \small
        Find $s$, $\theta$, $\phi$ minimizing\footnote{\textcite{magi} use a Bayesian approach.}
        $$
        \sum_i \|
         \hat x(t_i) - y_i
        \|
        $$
        subject to\footnote{Gradient matching. Integral matching can be used as well.} 
        $$
        \hat x'(t) = F(\hat x(t), t, \theta).
        $$
    \end{graphicbox}
\end{frame}
