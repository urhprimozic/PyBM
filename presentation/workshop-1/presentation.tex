\begin{frame}{ODE discovery with Probmot}
\begin{itemize}
    \item processes between entities (predator-prey models,...)
    \item build a list of possible models $[\vec{x}'(t)=\vec F_i(\vec{x}(t), t, \theta^{(i)})]_i$
    \item \textbf{estimate model parameters $\theta^{(i)}$}
    \item evaluate all possible models
\end{itemize}
\end{frame}
 \begin{frame}{Structure changes}
    \begin{itemize}
        \item migration from Java to Python
        \item gradient opt instead of EA 
        \item magnitutes faster parameter estimation
        \item conditional statemets support $x'(t) = \begin{cases}
            f(x,t, \theta) & \phi(x, t, \theta) > 0 \\
            g(x,t,\theta) & \text{otherwise}
        \end{cases}$
    \end{itemize}
 \end{frame}
 \begin{frame}{Other possible solutions}
    \begin{itemize}
        \item JIT compiled $F(x,t,\theta)$
        \item clever structure search 
        \item precise estimation just for best candidates
    \end{itemize}
 \end{frame}

\begin{frame}{Integration using multishooting\footfullcite{aydogmus2020modifiedmultipleshootingalgorithm}}
    \begin{itemize}
        \item integration: pick $x(t_0)$, solve ODE for $x_\theta(t)$, min discrepancy $\mathcal{L}(x_\theta)$ on data
        \item problem: caotic system
        \item solution: integrate shorter intervals, join trajectories   
    \end{itemize}
\end{frame}

\begin{frame}{Integration using multishooting\footfullcite{aydogmus2020modifiedmultipleshootingalgorithm}}
        \begin{graphicbox}[boxrule=2pt, title=Multishooting] 
        minimize discrepancy on data
        $$ \sum_i \mathcal{L}(x_i), \quad \text{where } x_i'=F(x_i,t,\theta), x_i(t_i)=s_i$$

        $$
    \end{graphicbox}
\end{frame}

\begin{frame}{Gradient matching for fully-observed systems}
    \begin{itemize}
        \item $\hat x(t)$ - interpolant of the data 
        \item interpolation: gaussian processes\footfullcite{wenk2019fastgaussianprocessbased}, cubic splines\footfullcite{varah-splines}
        %\item efective derivation
    \end{itemize}
    \begin{graphicbox}[boxrule=2pt, title=Least squares formulation]
       Find $
        \operatorname{argmin}\|\hat x'(t) - F(\hat x(t), t, \theta)  \|
        $
    \end{graphicbox}
\end{frame}

\begin{frame}{Collocation\footfullcite{magi}}
    \begin{itemize}
        \item $x\colon I \to R^n$ partially observed
        \item choose $l$ \emph{collocation} points $(\tau_1, s_1), (\tau_2, s_2), \cdots, (\tau_l, s_l) \in I\times R^n$  
        \item $\hat x(t)$: interpolant of $(\tau_i, s_i)_{i}$, {\color{gray} hyperparameters $\phi$}
        \item choose such $s=(s_1, \cdots, s_l), \phi$ that $\hat x(t)$ solves ODE and matches data 
    \end{itemize}
\end{frame}
\begin{frame}{Collocation}
%
     \begin{small}
        Let $ \hat x(t)$ interpolant of $(\tau_i, s_i)$ with hyperparameters $\phi$ and collocation points  $s=(s_1, \cdots, s_l)$. Let $(t_i, y_i)$ be a noisy set of measurments.
     \end{small}
 %    
    \begin{graphicbox}
        Find $s$, $\theta$, $phi$, minimizing\footnote{\cite{magi} use bayesian approach}
        $$
        ||
         \hat x(t_i) - y_i
        ||
        $$
        such that\footnote{Gradient matching. Integral matching can be used as well.} 
        $$
        \hat x'(t) = F(\hat x(t), t, \theta).
        $$
    \end{graphicbox}
\end{frame}
