\begin{frame}{ODE discovery with Probmot}
\begin{itemize}
    \item processes between entities (predator-prey models,...)
    \item build a list of possible models $[\vec{x}'(t)=\vec F_i(\vec{x}(t), t, \theta^{(i)})]_i$
    \item \textbf{estimate model parameters $\theta^{(i)}$}
    \item evaluate all possible models
\end{itemize}
\end{frame}
 \begin{frame}{Structure changes}
    \begin{itemize}
        \item migration from Java to Python
        \item gradient opt instead of EA 
        \item magnitutes faster parameter estimation
        \item conditional statemets support $x'(t) = \begin{cases}
            f(x,t, \theta) & \phi(x, t, \theta) > 0 \\
            g(x,t,\theta) & \text{otherwise}
        \end{cases}$
    \end{itemize}
 \end{frame}
 \begin{frame}{Other possible solutions}
    \begin{itemize}
        \item JIT compiled $F(x,t,\theta)$
        \item clever structure search 
        \item precise estimation just for best candidates
    \end{itemize}
 \end{frame}
 \begin{frame}{Relaxed conditionals}

 \begin{frame}{Parameter estimation}
    \begin{itemize}
        \item $x\colon [t_0, t_1] \to \mathbb{R}^n$
        \item $x_\text{obs} = (x_{o_1}, x_{o_2}, \dotsc, x_{o_{n_\text{obs}}}) \in \mathbb{R}^{n_\text{obs}}$
    \end{itemize}
    %Let $x\colon [t_0, t_1] \to \mathbb{R}^n$ and

    \hfill
    \begin{graphicbox}[boxrule=2pt] 
    Given 
    \begin{itemize}
        \item ODE $ x'(t) =  F( x(t), t, \theta)$
        \item  measurments $(t_i, x_\text{obs}(t_i) + \epsilon_i)$, $\epsilon_i \sim N(\mu, \sigma^2)$
    \end{itemize} 
    estimate $\theta$.
\end{graphicbox}
 \end{frame}
\begin{frame}{Integration using multishooting\footfullcite{aydogmus2020modifiedmultipleshootingalgorithm}}
    \begin{itemize}
        \item integration: pick $x(t_0)$, solve ODE for $x_\theta(t)$, min discrepancy $\mathcal{L}(x_\theta)$ on data
        \item problem: caotic system
        \item solution: integrate shorter intervals, join trajectories   
    \end{itemize}
\end{frame}

\begin{frame}{Integration using multishooting\footfullcite{aydogmus2020modifiedmultipleshootingalgorithm}}
        \begin{graphicbox}[boxrule=2pt, title=Multishooting] 
        minimize discrepancy on data
        $$ \sum_i \mathcal{L}(x_i), \quad \text{where } x_i'=F(x_i,t,\theta), x_i(t_i)=s_i$$
        on 
        $$
        x_i(t_{i+1}) = s_i \quad \forall i=0,1, \dots N_\text{subintervals}
        $$
    \end{graphicbox}
\end{frame}
